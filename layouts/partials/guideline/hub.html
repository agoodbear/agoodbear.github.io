-}} {{- range $t := $zh.topics -}} {{- $groups := slice -}} {{- range $g := $t.groups -}} {{- $items := slice -}} {{- range $e := $g.items -}} {{- $x := $e -}} {{- $o := index $emap $e.doc -}} {{- if
